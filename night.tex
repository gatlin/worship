\OrderTitle{Night Praise and Prayer}

\begin{Litany}
\L{O God, come to our assistance.}
\P{O Lord, hasten to help us.}
\L{The Lord Almighty grant us a restful night and peace at the last.}
\Amen
\end{Litany}

\begin{Litany}
\L{As you have made this day, O God, you also make the night.}
\L{Give light for our comfort.}
  \L{Come upon us with quietness and still our souls that we may}
\Indent\L{listen for the
  whisper of your Spirit and}
\Indent\L{be attentive to your nearness in our dreams.}
\L{Empower us to rise again in new life to proclaim your praise, and show
  Christ to the world.}
\Amen
\end{Litany}
